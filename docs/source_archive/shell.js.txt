const SHEET={open:false,onClose:null};
function openSheet(html,onClose,refresh){SHEET.open=true;SHEET.onClose=onClose||null;SHEET.refresh=refresh||null;
  $("sheet").innerHTML=`<div class="sheet-bg" onclick="if(event.target===this)closeSheet()"><div class="sheet"><div class="grab"></div>${html}</div></div>`;}

function closeSheet(){SHEET.open=false;SHEET.refresh=null;$("sheet").innerHTML="";const f=SHEET.onClose;SHEET.onClose=null;if(f)f();render();}

/* ===== Navigation ===== */
function navFor(r){
  if(r==="ct")return[["bord","📊","Tableau de bord"],["ordres","📋","Ordres"],["rotations","⏱️","Rotations"],["pointages","📦","Pointages"],["convois","🚚","Convois"],["incidents","🚨","Incidents"],["reglages","⚙️","Réglages"]];
  if(r==="agent")return agentSite()==="PSK"
    ?[["rotations","⏱️","Rotations"],["pointage","📦","Pointer"],["programme","📋","Programme"],["incidents","🚨","Incidents"]]
    :[["rotations","⏱️","Rotations"],["pointage","📦","Pointer"],["convois","🚚","Convois"],["programme","📋","Programme"],["incidents","🚨","Incidents"]];
  if(r==="hse")return[["convois","🚚","Convois"],["incidents","🚨","Incidents"]];
  if(r==="site")return[["reception","📥","Réception"],["programme","📋","Flux prévus"]];
  if(r==="presta")return[["flotte","🚚","Ma flotte"],["programme","📋","Ordres"],["rotations","⏱️","Rotations"],["export","⬇️","Export"]];
  if(r==="pointeur")return[["pointage","📦","Pointer"]];
  return[];}
function navBadge(id){const r=role(),d=S.day;
  if(id==="programme"&&r==="presta")return all("programmes").filter(p=>p.date===d&&p.prestataire===ent()&&!p.confirm&&!p.closed).length;
  if(id==="reception")return all("pointages").filter(p=>p.site===SITE_KEYS[ent()]&&!p.contreSigne).length;
  if(id==="flotte"&&r==="presta")return flotteDuJour(ent(),today())?0:1;
  return 0;}
function go(t){S.tab=t;S.ui.tab=t;LS.set("ui",S.ui);window.scrollTo(0,0);render();}

let _renderPending=false;
function fieldKey(el){if(!el||!/^(INPUT|SELECT|TEXTAREA)$/.test(el.tagName))return null;return el.id?"#"+el.id:["onchange","oninput","placeholder"].map(a=>el.getAttribute(a)).find(Boolean)||null;}
function refocus(key,sel){if(!key)return;let el=null;if(key[0]==="#")el=document.getElementById(key.slice(1));
  else el=[...document.querySelectorAll("input,select,textarea")].find(x=>["onchange","oninput","placeholder"].some(a=>x.getAttribute(a)===key));
  if(el){el.focus();try{if(sel!=null&&el.setSelectionRange)el.setSelectionRange(sel,sel);}catch(e){}}}
function renderKeep(){if(_renderPending)return;_renderPending=true;requestAnimationFrame(()=>{_renderPending=false;
  const a=document.activeElement;const key=fieldKey(a);let sel=null;try{sel=a&&a.selectionStart;}catch(e){}
  if(SHEET.open&&SHEET.refresh)SHEET.refresh();render();refocus(key,sel);});}
function render(){
  const root=$("root");
  if(!S.session){root.innerHTML=loginView();return;}
  const nav=navFor(role());if(!S.tab||!nav.find(n=>n[0]===S.tab))S.tab=(S.ui.tab&&nav.find(n=>n[0]===S.ui.tab))?S.ui.tab:nav[0][0];
  const main=viewFor(S.tab);
  const mob=nav.length>5?nav.slice(0,4).concat([["__plus","☰","Plus"]]):nav;
  root.innerHTML=`<div class="top"><div class="grow"><div class="brand">Transferts CT</div><div class="who">${esc(whoLabel())}</div></div>
    <button id="syncpill" class="syncpill" onclick="syncSheet()"></button><button class="iconbtn" onclick="accountSheet()">☰</button></div>
    <div class="layout"><nav class="side">${nav.map(([id,ic,lb])=>{const b=navBadge(id);return `<button class="${id===S.tab?"on":""}" onclick="go('${id}')"><span class="ic">${ic}</span>${lb}${b?`<span class="badge">${b}</span>`:""}</button>`;}).join("")}</nav>
    <main>${main}</main></div>
    <nav class="bottomnav">${mob.map(([id,ic,lb])=>{const b=id==="__plus"?0:navBadge(id);const on=id===S.tab||(id==="__plus"&&!mob.find(m=>m[0]===S.tab));
      return `<button class="${on?"on":""}" onclick="${id==="__plus"?"moreSheet()":`go('${id}')`}"><span class="ic">${ic}</span>${lb}${b?`<span class="badge">${b}</span>`:""}</button>`;}).join("")}</nav>`;
  document.body.classList.toggle("nonav",nav.length<2);
  paintSync();if(typeof afterRender==="function")afterRender();}

async function logout(){const n=pendingCount();
  if(n&&!confirm(n+" modification(s) ne sont pas encore envoyées et seront perdues. Se déconnecter quand même ?"))return;
  try{await rpc("tct_logout",{p_token:S.session.token});}catch(e){}
  clearLocal();closeSheet();}
function forceLogout(msg){S.session=null;LS.del("session");render();toast(msg,true);}
function clearLocal(){S.session=null;LS.del("session");TYPES.forEach(t=>{S.data[t]={};LS.del("d_"+t);});S.outbox={};S.lastSeq=0;S.seen={};persistMeta();LS.del("settings");S.tab=null;}

const LOGIN={step:"code",pin:"",busy:false,pending:null,mode:null};
function loginView(){const L=LOGIN;let h=`<div class="login"><div class="box">`;
  if(L.step==="code"){const last=LS.get("lastOper","");
    h+=`<h1>Transferts CT</h1><p>Conakry Terminal · Service Transport</p>
      <label class="f" style="color:#AEC4DA">Nom ou matricule</label><input class="in" id="lg_op" value="${esc(last)}" placeholder="ex. A. Diallo / 1666" autocomplete="off">
      <label class="f" style="color:#AEC4DA">Code d'accès</label>
      <div class="pindots">${Array.from({length:Math.max(4,L.pin.length)},(_,i)=>`<i class="${L.pin.length>i?"f":""}"></i>`).join("")}</div>
      <div class="pinpad">${[1,2,3,4,5,6,7,8,9].map(n=>`<button onclick="pinKey('${n}')">${n}</button>`).join("")}<button onclick="pinKey('C')" style="font-size:15px">Effacer</button><button onclick="pinKey('0')">0</button><button onclick="doLogin()" style="background:#F2B84B;color:#1A2430;font-size:16px">${L.busy?"…":"Entrer"}</button></div>
      <p class="small center" style="margin-top:14px">Code personnel communiqué par le service Transport.</p>`;}
  else{const P=L.pending;const back=`<button class="back" onclick="cancelChoice()">‹ Changer de code</button>`;
    if(P.role==="agent")h+=back+`<h1>Bonjour ${esc(P.operateur)}</h1><p>Sur quel site travailles-tu ?</p><div class="entgrid">${["CT","PSK"].map(e=>`<div class="rolecard" onclick="finishLogin('agent','${e}','')">${esc(AGENT_SITE_LABEL[e])}</div>`).join("")}</div>`;
    else if(P.role==="site")h+=back+`<h1>Bonjour ${esc(P.operateur)}</h1><p>Quel est ton site ?</p><div class="entgrid">${Object.keys(SITE_KEYS).map(e=>`<div class="rolecard" onclick="finishLogin('site','${e}','')">${esc(SITE_KEYS[e])}</div>`).join("")}</div>`;
    else if(P.role==="presta"&&L.mode!=="pointage")h+=back+`<h1>${esc(P.entity)}</h1><p>Que veux-tu faire ?</p>
      <div class="rolecard" onclick="finishLogin('presta','${esc(P.entity)}','')"><div class="ri">🚚</div><div><div class="rt">Flotte, ordres et rotations</div><div class="rd">Déclarer la flotte, confirmer les ordres, suivre les camions.</div></div></div>
      <div class="rolecard" onclick="LOGIN.mode='pointage';render()"><div class="ri">📦</div><div><div class="rt">Pointage des conteneurs</div><div class="rd">Pointer les arrivées et départs sur un site.</div></div></div>`;
    else h+=`<button class="back" onclick="LOGIN.mode=null;render()">‹ Retour</button><h1>Pointage ${esc(P.entity)}</h1><p>Sur quel site pointes-tu ?</p>
      ${SITES.map(s=>`<div class="rolecard" onclick="finishLogin('pointeur','${esc(P.entity)}','${esc(s)}')"><div class="ri">📍</div><div class="rt">${esc(s)}</div></div>`).join("")}`;}
  return h+`</div></div>`;}
function pinKey(k){const op=$("lg_op");const opv=op?op.value:"";if(k==="C")LOGIN.pin=LOGIN.pin.slice(0,-1);else if(LOGIN.pin.length<8)LOGIN.pin+=k;
  render();const o=$("lg_op");if(o)o.value=opv;}
document.addEventListener("keydown",e=>{if(S.session||LOGIN.step!=="code")return;if(document.activeElement&&document.activeElement.id==="lg_op"){if(e.key==="Enter")doLogin();return;}
  if(/^[0-9]$/.test(e.key))pinKey(e.key);else if(e.key==="Backspace")pinKey("C");else if(e.key==="Enter")doLogin();});
async function doLogin(){if(LOGIN.busy)return;const op=(($("lg_op")||{}).value||"").trim();
  if(!op){toast("Indique ton nom ou matricule",true);return;}if(LOGIN.pin.length<4){toast("Saisis ton code",true);return;}
  LOGIN.busy=true;render();$("lg_op").value=op;
  try{const s=await rpc("tct_connect",{p_code:LOGIN.pin,p_operateur:op});LS.set("lastOper",op);LOGIN.pin="";LOGIN.busy=false;
    if(s.role==="ct"||s.role==="hse"){enterSession(s);return;}
    LOGIN.pending=s;LOGIN.mode=null;LOGIN.step="choice";render();}
  catch(e){LOGIN.busy=false;LOGIN.pin="";render();const o=$("lg_op");if(o)o.value=op;toast(errMsg(e),true);}}
async function finishLogin(r,e,site){const P=LOGIN.pending;if(!P||LOGIN.busy)return;LOGIN.busy=true;
  try{const s=await rpc("tct_session_set",{p_token:P.token,p_role:r,p_entity:e,p_site:site});enterSession(s);}
  catch(err){LOGIN.busy=false;if(err.message==="session_expiree"){cancelChoice();}toast(errMsg(err),true);}}
function cancelChoice(){LOGIN.pending=null;LOGIN.mode=null;LOGIN.step="code";LOGIN.busy=false;render();}
function enterSession(s){clearLocal();S.session=s;LS.set("session",s);Object.assign(LOGIN,{step:"code",pin:"",busy:false,pending:null,mode:null});S.day=today();
  render();toast("Bienvenue "+s.operateur);syncNow();}
